# C-Tests brauchen eine libc für die Ausgabe. Gebaut werden sie überall,
# wo es eine gibt; als ctest registriert nur dort, wo sie hier laufen können.
# Xernel-Tests startet ./nhal build --xernel --test in QEMU.
set(NHAL_TESTS_RUN_HERE ON)
if(CMAKE_CROSSCOMPILING AND NOT CMAKE_CROSSCOMPILING_EMULATOR)
    set(NHAL_TESTS_RUN_HERE OFF)
endif()

if(NOT NHAL_PLATFORM_FREESTANDING)
    function(nhal_c_test name source)
        add_executable(${name} ${source})
        target_link_libraries(${name} PRIVATE nhal::nhal)
        target_include_directories(${name} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
        set_target_properties(${name} PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON)
        set_property(GLOBAL APPEND PROPERTY NHAL_C_TEST_NAMES ${name})
        if(NHAL_TESTS_RUN_HERE)
            add_test(NAME ${name} COMMAND ${name} WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}")
        else()
            # Für Läufer auf dem Zielsystem (z. B. tools/xernel_qemu.py).
            file(APPEND "${NHAL_TARGET_TESTS_FILE}" "$<TARGET_FILE:${name}>\n")
        endif()
    endfunction()
    set(NHAL_TARGET_TESTS_FILE "${CMAKE_BINARY_DIR}/nhal_target_tests.in")
    file(WRITE "${NHAL_TARGET_TESTS_FILE}" "")

    nhal_c_test(core core/core_test.c)
    if("mem.pool" IN_LIST NHAL_SELECTED_FEATURES AND "mem.frame" IN_LIST NHAL_SELECTED_FEATURES)
        nhal_c_test(mem features/mem_test.c)
    endif()
    if("fs.file" IN_LIST NHAL_SELECTED_FEATURES)
        nhal_c_test(fs features/fs_test.c)
    endif()
    if("thread.scheduler" IN_LIST NHAL_SELECTED_FEATURES)
        nhal_c_test(scheduler features/scheduler_test.c)
    endif()
    if("net.socket_udp" IN_LIST NHAL_SELECTED_FEATURES)
        nhal_c_test(net features/net_test.c)
    endif()
    if("audio.output" IN_LIST NHAL_SELECTED_FEATURES)
        nhal_c_test(audio features/audio_test.c)
    endif()
    if("input.gamepad" IN_LIST NHAL_SELECTED_FEATURES)
        nhal_c_test(gamepad features/gamepad_test.c)
    endif()
    if("gfx.device" IN_LIST NHAL_SELECTED_FEATURES)
        nhal_c_test(gfx features/gfx_test.c)
        nhal_c_test(gfx2 features/gfx2_test.c)
    endif()
    if("input.state" IN_LIST NHAL_SELECTED_FEATURES AND "input.actions" IN_LIST NHAL_SELECTED_FEATURES)
        nhal_c_test(input features/input_test.c)
    endif()
    if("sys.shell" IN_LIST NHAL_SELECTED_FEATURES)
        nhal_c_test(shell features/shell_test.c)
    endif()
    if("window.basic" IN_LIST NHAL_SELECTED_FEATURES)
        nhal_c_test(window features/window_test.c)
    endif()
    # Zusammenspiel: Fenster → Oberfläche → Eingabe → Aktionen (examples/game_input im Selbsttest).
    if("window.basic" IN_LIST NHAL_SELECTED_FEATURES AND "gfx.device" IN_LIST NHAL_SELECTED_FEATURES AND
       "input.state" IN_LIST NHAL_SELECTED_FEATURES AND "input.actions" IN_LIST NHAL_SELECTED_FEATURES)
        nhal_c_test(game_input "${NHAL_ROOT}/examples/game_input/main.c")
        target_compile_definitions(game_input PRIVATE NHAL_DEMO_SELFTEST=1)
    endif()
    if("gfx.mesh_shader" IN_LIST NHAL_SELECTED_FEATURES)
        nhal_c_test(mesh features/mesh_test.c)
    endif()
    if("gfx.raytracing" IN_LIST NHAL_SELECTED_FEATURES)
        nhal_c_test(raytracing features/raytracing_test.c)
    endif()
    if(NHAL_PLATFORM STREQUAL "web")
        # Für den Lauf im Browser (WebGPU, Web Audio, Gamepad gibt es in node nicht).
        configure_file("${CMAKE_CURRENT_SOURCE_DIR}/web/runner.html" "${CMAKE_CURRENT_BINARY_DIR}/runner.html" COPYONLY)
        get_property(_nhal_names GLOBAL PROPERTY NHAL_C_TEST_NAMES)
        string(REPLACE ";" "\n" _nhal_names "${_nhal_names}")
        file(WRITE "${CMAKE_BINARY_DIR}/nhal_browser_tests.txt" "${_nhal_names}\n")
    endif()
    if(NOT NHAL_TESTS_RUN_HERE)
        file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/nhal_target_tests.txt" INPUT "${NHAL_TARGET_TESTS_FILE}")
    endif()

endif()

add_test(NAME nhalgen_unit

add_test(NAME catalog_check

         COMMAND "${Python3_EXECUTABLE}" "${NHAL_ROOT}/tools/nhalgen/nhalgen.py" matrix --check)

         COMMAND "${Python3_EXECUTABLE}" "${NHAL_ROOT}/tools/guard/guard.py" --root "${NHAL_ROOT}")
